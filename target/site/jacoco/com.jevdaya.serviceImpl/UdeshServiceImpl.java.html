package com.jevdaya.serviceImpl;

import com.jevdaya.Entity.Udesh;
import com.jevdaya.repo.UdeshRepo;
import com.jevdaya.service.UdeshService;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;

@Service
public class UdeshServiceImpl implements UdeshService {

    private final UdeshRepo repo;

    // Constructor injection (preferred over @Autowired field)
    public UdeshServiceImpl(UdeshRepo repo) {
        this.repo = repo;
    }

    @Override
    public List<Udesh> getAllUdesh() {
        return repo.findAll();
    }

    @Override
    public Udesh getById(int id) {
        return repo.findById(id).orElse(null);
    }

    @Override
    public Udesh addUdesh(Udesh udesh) {
        return repo.save(udesh);
    }

    @Override
    public Udesh updateUdesh(int id, Udesh udesh) {
        Optional<Udesh> existingOpt = repo.findById(id);

        if (existingOpt.isPresent()) {
            Udesh existing = existingOpt.get();
            existing.setDescription(udesh.getDescription());
            // You can add more fields here if Udesh has them
            return repo.save(existing);
        }
        return null;
    }

    @Override
    public void deleteUdesh(int id) {
        repo.deleteById(id);
    }
}
