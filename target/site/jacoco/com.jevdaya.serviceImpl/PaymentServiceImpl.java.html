package com.jevdaya.serviceImpl;

import org.json.JSONObject;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import com.jevdaya.Entity.Payment;
import com.jevdaya.repo.PaymentRepository;
import com.jevdaya.service.PaymentService;
import com.razorpay.Order;
import com.razorpay.RazorpayClient;
import com.razorpay.Utils;

@Service
public class PaymentServiceImpl implements PaymentService {

    private static final String KEY_ID = "rzp_test_SahHHcqfDLxNc9";
    private static final String KEY_SECRET = "6U69PBnxMGffotooyDPcF5rK";

    @Autowired
    private PaymentRepository repo;

    @Override
    public String createOrder(double amount) {
        try {
            RazorpayClient client = new RazorpayClient(KEY_ID, KEY_SECRET);

            JSONObject options = new JSONObject();
            options.put("amount", (int)(amount * 100)); // in paise
            options.put("currency", "INR");
            options.put("receipt", "txn_123456");

            Order order = client.orders.create(options);
            return order.toString();

        } catch (Exception e) {
            e.printStackTrace();
            return null;
        }
    }

    @Override
    public String verifyPayment(String orderId, String paymentId, String signature) {
        try {
            String generatedSignature = Utils.getHash(orderId + "|" + paymentId, KEY_SECRET);

            if (generatedSignature.equals(signature)) {
                Payment p = new Payment();
                p.setOrderId(orderId);
                p.setPaymentId(paymentId);
                p.setStatus("SUCCESS");
                repo.save(p);

                return "Payment Verified";
            } else {
                return "Invalid Signature";
            }

        } catch (Exception e) {
            return "Error";
        }
    }
}
