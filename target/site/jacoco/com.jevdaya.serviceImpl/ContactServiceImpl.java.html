package com.jevdaya.serviceImpl;

import com.jevdaya.Entity.Contact;
import com.jevdaya.repo.ContactRepository;
import com.jevdaya.service.ContactService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;
import java.util.Optional;

@Service
public class ContactServiceImpl implements ContactService {

    @Autowired
    private ContactRepository contactRepository;

    private String pageTitle = "Contact Us";
    private String pageSubTitle = "We'd love to hear from you";
    private String pageDescription = "<p>Feel free to reach out for any queries or support.</p>";
    @Override
    public Contact saveMessage(Contact contact) {
        return contactRepository.save(contact);
    }

    @Override
    public List<Contact> getAllMessages() {
        return contactRepository.findAll();
    }

    @Override
    public Contact getMessageById(Long id) {
        Optional<Contact> contact = contactRepository.findById(id);
        return contact.orElse(null);
    }

    @Override
    public Contact updateMessage(Long id, Contact updatedContact) {
        Optional<Contact> existingOpt = contactRepository.findById(id);
        if (existingOpt.isPresent()) {
            Contact existing = existingOpt.get();
            existing.setName(updatedContact.getName());
            existing.setEmail(updatedContact.getEmail());
            existing.setSubject(updatedContact.getSubject());
            existing.setMessage(updatedContact.getMessage());
            // Add more fields if you have (e.g., phone, createdAt etc.)
            return contactRepository.save(existing);
        }
        return null;
    }

    @Override
    public boolean deleteMessage(Long id) {
        if (contactRepository.existsById(id)) {
            contactRepository.deleteById(id);
            return true;
        }
        return false;
    }
    

    @Override
    public Map<String, String> getPageContent() {
        return Map.of(
            "title", pageTitle,
            "subTitle", pageSubTitle,
            "description", pageDescription
        );
    }

    @Override
    public void updatePageContent(String title, String subTitle, String description) {
        if (title != null && !title.trim().isEmpty()) {
            this.pageTitle = title;
        }
        if (subTitle != null) {
            this.pageSubTitle = subTitle;
        }
        if (description != null) {
            this.pageDescription = description;
        }
        System.out.println("Page content updated - Title: " + pageTitle); // for debugging
    }
}
